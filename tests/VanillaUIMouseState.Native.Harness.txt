using System;
using System.IO;
using System.Reflection;
using System.Runtime.Serialization;

public static class VanillaUIMouseStateNative
{
    private static readonly BindingFlags Fields = BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.Static | BindingFlags.Instance;
    private static Type _main, _playerType, _renderer, _manager, _mod;
    private static object _player;
    private static int _checks, _failed;

    public static int Main(string[] args)
    {
        try
        {
            if (IntPtr.Size != 4) throw new Exception("Requires actual net48 x86 process");
            string game = "D:/App/Game/Steam/steamapps/common/Terraria";
            Assembly terraria = Assembly.LoadFrom(Path.Combine(game, "Terraria.exe"));
            AppDomain.CurrentDomain.AssemblyResolve += (sender, request) =>
            {
                string name = new AssemblyName(request.Name).Name;
                if (name == "Terraria") return terraria;
                foreach (string path in new[] {
                    Path.Combine(AppDomain.CurrentDomain.BaseDirectory, name + ".dll"),
                    Path.Combine(game, name + ".dll"),
                    Path.Combine(game, "TerrariaModder/core", name + ".dll"),
                    Path.Combine(game, "TerrariaModder/core/deps", name + ".dll"),
                    Path.Combine(Environment.GetEnvironmentVariable("QOT_SOURCE_ROOT"), "lib", name + ".dll") })
                    if (File.Exists(path)) return Assembly.LoadFrom(path);
                foreach (string resource in terraria.GetManifestResourceNames())
                    if (resource.EndsWith("." + name + ".dll", StringComparison.OrdinalIgnoreCase))
                        using (var stream = terraria.GetManifestResourceStream(resource))
                        using (var buffer = new MemoryStream())
                        { stream.CopyTo(buffer); return Assembly.Load(buffer.ToArray()); }
                return null;
            };
            Set(terraria.GetType("Terraria.Program", true), "SavePath", Path.Combine(AppDomain.CurrentDomain.BaseDirectory, "test-save"));
            _main = terraria.GetType("Terraria.Main", true);
            _playerType = terraria.GetType("Terraria.Player", true);
            _player = FormatterServices.GetUninitializedObject(_playerType);
            Array players = Array.CreateInstance(_playerType, 1);
            players.SetValue(_player, 0);
            Set(_main, "player", players); Set(_main, "myPlayer", 0);
            Set(_main, "gameMenu", false); Set(_main, "dedServ", false); Set(_main, "netMode", 0);
            Set(_main, "playerInventory", true); Set(_playerType, "active", true, _player);

            Assembly core = Assembly.LoadFrom(Path.Combine(game, "TerrariaModder/core/TerrariaModder.Core.dll"));
            _renderer = core.GetType("TerrariaModder.Core.UI.UIRenderer", true);
            SetMouse(true, true);
            _renderer.GetMethod("UnregisterPanelBounds", Fields).Invoke(null, new object[] { "native-test.never-registered" });
            Check(!MouseInterface && !BlockMouse, "真实 Core 注销不存在的面板也会清除已有鼠标阻挡状态");

            Assembly production = Assembly.LoadFrom(Path.GetFullPath(args[0]));
            _mod = production.GetType("QualityOfTerraria.Mod", true);
            _mod.GetProperty("WorldActive", Fields).GetSetMethod(true).Invoke(null, new object[] { true });
            _manager = production.GetType("QualityOfTerraria.UI.UIManager", true);
            // Represent an already initialized world; all Update bodies remain untouched.
            // Bypass initial registration/persistence, which is unrelated to per-frame input ownership.
            Set(_manager, "_initialized", true);
            Check((bool)_manager.GetProperty("CanRender", Fields).GetValue(null), "真实世界内 UIManager.CanRender 已就绪");
            foreach (bool mouse in new[] { false, true })
                foreach (bool block in new[] { false, true })
                {
                    SetMouse(mouse, block);
                    _manager.GetMethod("UpdateAll", Fields).Invoke(null, null);
                    Console.WriteLine("UpdateAll initial mouseInterface=" + mouse + " blockMouse=" + block +
                        " => mouseInterface=" + MouseInterface + " blockMouse=" + BlockMouse);
                    Check(MouseInterface == mouse && BlockMouse == block,
                        "全部模组窗口关闭时保留原版输入状态 mouseInterface=" + mouse + " blockMouse=" + block);
                }
            Type stats = production.GetType("QualityOfTerraria.UI.PlayerStatsUI", true);
            string statsPanel = (string)stats.GetField("PanelId", Fields).GetValue(null);
            _renderer.GetMethod("RegisterPanelBounds", Fields, null,
                new[] { typeof(string), typeof(int), typeof(int), typeof(int), typeof(int) }, null)
                .Invoke(null, new object[] { statsPanel, 0, 0, 100, 100 });
            Check((bool)_renderer.GetProperty("IsBlocking", Fields).GetValue(null), "真实 Core 已注册本模组模态面板");
            SetMouse(true, true);
            _manager.GetMethod("UpdateAll", Fields).Invoke(null, null);
            Check(!MouseInterface && !BlockMouse && !(bool)_renderer.GetProperty("IsBlocking", Fields).GetValue(null),
                "本模组最后一个模态面板关闭时正确解除自身阻挡");

            Type travelShop = production.GetType("QualityOfTerraria.UI.TravelShopRefreshUI", true);
            Action draw = (Action)Delegate.CreateDelegate(typeof(Action), travelShop.GetMethod("Draw", Fields));
            _manager.GetMethod("RegisterDraw", Fields).Invoke(null, new object[] { "native-test.closed-travel-shop", draw });
            SetMouse(true, true);
            var callbacks = (System.Collections.Generic.IDictionary<string, Action>)
                _renderer.GetField("_panelDrawCallbacks", Fields).GetValue(null);
            callbacks["native-test.closed-travel-shop"]();
            Check(MouseInterface && BlockMouse,
                "真实 Core 已注册的关闭旅商绘制回调保留原版背包悬停状态");
            _renderer.GetMethod("UnregisterPanelDraw", Fields).Invoke(null, new object[] { "native-test.closed-travel-shop" });

            Set(_main, "gameMenu", true);
            SetMouse(true, true);
            _manager.GetMethod("UpdateAll", Fields).Invoke(null, null);
            Check(!MouseInterface && !BlockMouse, "返回菜单的 ResetSession 仍清除本次会话鼠标阻挡");
            Console.WriteLine("真实 net48 x86 鼠标状态检查：" + (_checks - _failed) + "/" + _checks + "（未启动游戏）");
            return _failed;
        }
        catch (Exception error) { Console.WriteLine(error); return 1; }
    }

    private static bool MouseInterface => (bool)_playerType.GetField("mouseInterface", Fields).GetValue(_player);
    private static bool BlockMouse => (bool)_main.GetField("blockMouse", Fields).GetValue(null);
    private static void SetMouse(bool mouse, bool block)
    { Set(_playerType, "mouseInterface", mouse, _player); Set(_main, "blockMouse", block); }
    private static void Set(Type type, string name, object value, object target = null)
    { type.GetField(name, Fields).SetValue(target, value); }
    private static void Check(bool condition, string message)
    { _checks++; if (!condition) _failed++; Console.WriteLine((condition ? "PASS " : "FAIL ") + message); }
}
