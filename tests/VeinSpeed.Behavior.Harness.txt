using System;
using System.IO;
using System.Reflection;
using System.Collections.Generic;
using System.Linq;
using QualityOfTerraria;
using QualityOfTerraria.Patches;
using Terraria;
using TerrariaModder.Core;
using TerrariaModder.Core.Config;
public static class VeinSpeedChecks {
 static int failures,checks;
 static readonly BindingFlags Flags=BindingFlags.Static|BindingFlags.Instance|BindingFlags.Public|BindingFlags.NonPublic;
 static readonly string[] Keys={"VeinMiningEnabled","VeinMiningAutoLoot","VeinMiningRange","VeinMiningMaxBlocks","VeinMiningCheckPickaxePower","VeinMiningMatchExactType","VeinMiningShowMessages","DigSpeedBoost","AxeSpeedBoost","PlaceSpeedBoost"};
 static void Check(bool ok,string name){checks++;if(!ok)failures++;Console.WriteLine((ok?"PASS ":"FAIL ")+name);}
 static void Set(QualityOfTerrariaConfig c,string key,object value){c.GetType().GetProperty(key).SetValue(c,value);}
 static int Boost(QualityOfTerrariaConfig c,string key){return (int)c.GetType().GetProperty(key).GetValue(c);}
 static QualityOfTerrariaConfig Load(string directory,string json){Directory.CreateDirectory(directory);string file=Path.Combine(directory,"quality-of-terraria.client.json");if(json!=null)File.WriteAllText(file,json);return (QualityOfTerrariaConfig)ConfigManager.LoadForMod("quality-of-terraria",typeof(QualityOfTerrariaConfig).Assembly,directory,null,null);}
 static void Import(QualityOfTerrariaConfig config){var context=(ModContext)System.Runtime.Serialization.FormatterServices.GetUninitializedObject(typeof(ModContext));typeof(ModContext).GetField("<Config>k__BackingField",Flags).SetValue(context,config);typeof(QualityOfTerrariaConfig).Assembly.GetType("QualityOfTerraria.Features.VeinConfigMigration").GetMethod("Apply",Flags).Invoke(null,new object[]{context});}
 static void Speeds(int dig,int axe,int place){Set(Mod.Config,"DigSpeedBoost",dig);Set(Mod.Config,"AxeSpeedBoost",axe);Set(Mod.Config,"PlaceSpeedBoost",place);}
 public static int Main(){
  Check(new QualityOfTerrariaConfig().Version==3,"配置版本3");
  foreach(string key in Keys){var p=typeof(QualityOfTerrariaConfig).GetProperty(key);Check(p!=null&&Attribute.IsDefined(p,typeof(ClientAttribute)),key+" 为个人配置");}
  if(failures>0)return failures;
  Check(typeof(QualityOfTerrariaConfig).GetProperty("FastToolSpeed")==null,"旧固定速度不再暴露可见设置");
  var speed=typeof(QualityOfTerrariaConfig).Assembly.GetType("QualityOfTerraria.Features.ToolSpeedSystem");Check(speed!=null,"统一工具速度系统存在");if(speed==null)return failures;
  var multiplier=speed.GetMethod("GetMultiplier",Flags);
  foreach(int boost in new[]{-100,0,1,2,3,100})Check(Convert.ToSingle(multiplier.Invoke(null,new object[]{boost}))==1+Math.Max(0,Math.Min(3,boost)),"倍率边界 "+boost);
  var defaults=new QualityOfTerrariaConfig();Check(Boost(defaults,"DigSpeedBoost")==2&&Boost(defaults,"AxeSpeedBoost")==1&&Boost(defaults,"PlaceSpeedBoost")==0,"新安装默认挖掘3倍/砍伐2倍/放置原版");
  foreach(string key in Keys){var p=defaults.GetType().GetProperty(key);var range=(RangeAttribute)Attribute.GetCustomAttribute(p,typeof(RangeAttribute));if(p.PropertyType==typeof(int)){int min=key=="VeinMiningRange"||key=="VeinMiningMaxBlocks"?1:0;int max=key=="VeinMiningRange"?30:key=="VeinMiningMaxBlocks"?500:3;Check(range!=null&&range.Min==min&&range.Max==max,key+" 声明范围");}}
  string folder=Path.Combine(Environment.CurrentDirectory,"bin","vein-speed","migration-"+Guid.NewGuid().ToString("N"));
  foreach(bool old in new[]{false,true}){var c=Load(Path.Combine(folder,old.ToString()),"{\"_version\":2,\"FastToolSpeed\":"+old.ToString().ToLowerInvariant()+"}");Check(Boost(c,"DigSpeedBoost")== (old?1:0)&&Boost(c,"AxeSpeedBoost")== (old?1:0)&&Boost(c,"PlaceSpeedBoost")== (old?1:0),"真实Core旧FastToolSpeed "+old+" 迁移");}
  string importFolder=Path.Combine(folder,"import");Directory.CreateDirectory(importFolder);string oldFile=Path.Combine(importFolder,"vein-miner.client.json");File.WriteAllText(oldFile,"{\"AutoLoot\":false,\"VeinMiningEnabled\":false,\"DigSpeedBoost\":3,\"AxeSpeedBoost\":0,\"PlaceSpeedBoost\":0,\"VeinMiningRange\":19,\"VeinMiningMaxBlocks\":81,\"CheckPickaxePower\":false,\"MatchExactTileType\":false,\"ShowMessages\":false}");
  var imported=Load(importFolder,"{\"_version\":2,\"FastToolSpeed\":true}");Import(imported);Check(Boost(imported,"DigSpeedBoost")==3&&Boost(imported,"AxeSpeedBoost")==0&&Boost(imported,"PlaceSpeedBoost")==0,"独立插件实际速度覆盖旧固定倍数迁移");Check(!(bool)imported.GetType().GetProperty("VeinMiningAutoLoot").GetValue(imported)&&Boost(imported,"VeinMiningRange")==19&&Boost(imported,"VeinMiningMaxBlocks")==81,"导入旧自动收纳/范围/数量");Check(Keys.Where(k=>imported.GetType().GetProperty(k).PropertyType==typeof(bool)).All(k=>!(bool)imported.GetType().GetProperty(k).GetValue(imported)),"导入旧开关别名");
  Set(imported,"DigSpeedBoost",1);imported.Save();Import(imported);Check(Boost(imported,"DigSpeedBoost")==1,"同进程不重复导入覆盖用户设置");var reloaded=Load(importFolder,null);Import(reloaded);Check(Boost(reloaded,"DigSpeedBoost")==1,"重启不重复导入覆盖新设置");Check(File.Exists(oldFile),"迁移保留旧插件配置文件");
  var fresh=Load(Path.Combine(folder,"fresh"),null);File.Copy(oldFile,Path.Combine(Path.GetDirectoryName(fresh.FilePath),"vein-miner.client.json"));Import(fresh);Check(Boost(fresh,"DigSpeedBoost")==3,"真实Core首次创建新配置仍导入旧插件");
  var rollback=Load(Path.Combine(folder,"rollback"),"{\"_version\":2,\"FastToolSpeed\":true}");File.Copy(oldFile,Path.Combine(Path.GetDirectoryName(rollback.FilePath),"vein-miner.client.json"));QualityOfTerraria.Networking.SettingsSync.SaveSucceeds=false;Import(rollback);Check(Boost(rollback,"DigSpeedBoost")==1&&Boost(rollback,"AxeSpeedBoost")==1&&(bool)rollback.GetType().GetProperty("VeinMiningAutoLoot").GetValue(rollback),"保存核验失败回滚全部导入值");QualityOfTerraria.Networking.SettingsSync.SaveSucceeds=true;var retry=Load(Path.Combine(folder,"rollback"),null);Import(retry);Check(Boost(retry,"DigSpeedBoost")==3,"保存失败后重启仍可重试导入");Import(rollback);Check(Boost(rollback,"DigSpeedBoost")==3,"失败后可重试且成功才标记完成");
  Mod.Config=new QualityOfTerrariaConfig();Mod.WorldActive=true;Terraria.Main.myPlayer=0;var actor=new Player();
  for(int boost=0;boost<=3;boost++){int mult=boost+1;Speeds(boost,boost,boost);actor.pickSpeed=.8f;actor.tileSpeed=.5f;actor.wallSpeed=.8f;PlayerQoLPatches.ToolStatsPatch.Postfix(actor);Check(Math.Abs(actor.pickSpeed-.8f/mult)<.0001f&&Math.Abs(actor.tileSpeed-.5f/mult)<.0001f&&Math.Abs(actor.wallSpeed-.8f/mult)<.0001f,"原版统计之后只应用一次倍率 "+boost);
   foreach(var item in new[]{new Item{pick=50},new Item{axe=10},new Item{hammer=10},new Item{pick=50,axe=10},new Item{createTile=1},new Item{createWall=1},new Item{type=99}}){actor.itemAnimation=actor.itemAnimationMax=48;actor.itemTime=actor.itemTimeMax=48;PlayerQoLPatches.ToolAnimationPatch.Postfix(actor,item);PlayerQoLPatches.InstantRecallPatch.Postfix(actor,item);int animation=item.pick>0||item.axe>0||item.hammer>0?48/mult:48;int time=item.pick==0&&(item.axe>0||item.hammer>0)?48/mult:48;Check(actor.itemAnimation==animation&&actor.itemAnimationMax==animation&&actor.itemTime==time,"镐/斧/锤/复合工具/放置/武器不双算 "+boost+"/"+item.pick+"/"+item.axe+"/"+item.hammer+"/"+item.createTile+"/"+item.createWall);Check(item.useTime==48&&item.useAnimation==48,"不永久修改Item计时");}}
  Speeds(3,0,2);var axeItem=new Item{axe=10};actor.itemAnimation=actor.itemAnimationMax=48;actor.itemTime=48;PlayerQoLPatches.ToolAnimationPatch.Postfix(actor,axeItem);PlayerQoLPatches.InstantRecallPatch.Postfix(actor,axeItem);Check(actor.itemAnimation==48&&actor.itemTime==48,"三速度项相互独立");
  speed.GetMethod("Toggle",Flags).Invoke(null,null);actor.pickSpeed=1;PlayerQoLPatches.ToolStatsPatch.Postfix(actor);Check(actor.pickSpeed==1,"速度总开关关闭后保留原版");speed.GetMethod("ResetSession",Flags).Invoke(null,null);actor.pickSpeed=1;PlayerQoLPatches.ToolStatsPatch.Postfix(actor);Check(actor.pickSpeed==.25f,"返回菜单清理临时开关");Mod.VeinMinerLegacyLoaded=true;actor.pickSpeed=1;PlayerQoLPatches.ToolStatsPatch.Postfix(actor);Check(actor.pickSpeed==1,"旧插件仍加载时暂停重复速度");Mod.VeinMinerLegacyLoaded=false;
  Terraria.Main.netMode=2;Terraria.Main.myPlayer=255;QualityOfTerraria.Networking.SettingsSync.Personal=null;actor.pickSpeed=1;PlayerQoLPatches.ToolStatsPatch.Postfix(actor);Check(actor.pickSpeed==1,"未握手个人配置不套服务器默认速度");Terraria.Main.myPlayer=0;QualityOfTerraria.Networking.SettingsSync.Personal=new QualityOfTerrariaConfig();Set(QualityOfTerraria.Networking.SettingsSync.Personal,"DigSpeedBoost",0);actor.pickSpeed=1;PlayerQoLPatches.ToolStatsPatch.Postfix(actor);Check(actor.pickSpeed==1,"本地guard仍按确认个人配置读取");
  Console.WriteLine("连锁配置与速度行为检查 "+(checks-failures)+"/"+checks);return failures;
 }
}
namespace QualityOfTerraria{public static class Mod{public static QualityOfTerrariaConfig Config;public static bool WorldActive,VeinMinerLegacyLoaded;}}
namespace QualityOfTerraria.Networking{internal static class SettingsSync{internal static bool SaveSucceeds=true;internal static QualityOfTerrariaConfig Personal;internal static QualityOfTerrariaConfig GetPlayerConfig(int player){return Terraria.Main.netMode==2?Personal:Mod.Config;}internal static bool SaveVerified(ModConfig config,out string error){error=SaveSucceeds?null:"模拟保存失败";if(SaveSucceeds){config.Save();var raw=ConfigManager.ParseJsonRaw(File.ReadAllText(config.FilePath));return Convert.ToInt32(raw["_version"])==config.Version&&config.GetPropertyMetadata().Where(p=>p.Scope==ConfigManager.GetCurrentScope()).All(p=>raw.ContainsKey(p.Key)&&Equals(Convert.ChangeType(raw[p.Key],p.PropertyType),p.GetValue(config)));}return false;}}}
