namespace QualityOfTerraria {
public static class StorageRegression {
    static void Check(bool value, string message) { if (!value) throw new Exception(message); }
    public static void Run(string folder) {
        string path = Path.Combine(folder, "fixture.dat");
        StoragePersistence.Write(path, w => w.Write(10));
        StoragePersistence.Write(path, w => w.Write(20));
        int value;
        Check(StoragePersistence.Read(path, r => r.ReadInt32(), out value) && value == 20, "正常读写");
        try { StoragePersistence.Write(path, w => { w.Write(30); throw new IOException("中断样本"); }); } catch(IOException) {}
        Check(StoragePersistence.Read(path, r => r.ReadInt32(), out value) && value == 20, "中断保存保留主文件");
        Check(!File.Exists(path + ".tmp"), "清除临时文件");
        File.WriteAllBytes(path, new byte[] { 1 });
        Check(StoragePersistence.Read(path, r => r.ReadInt32(), out value) && value == 10, "备份恢复");
        Check(StoragePersistence.Read(path, r => r.ReadInt32(), out value) && value == 10, "修复主文件");
        Check(Mod.Errors > 0, "损坏存档记录日志");
        Check(StoragePersistence.Identity("local:C:/Players/A.plr") != StoragePersistence.Identity("local:D:/Players/A.plr"), "同名角色隔离");
        Check(StoragePersistence.Identity("guid:world-1") != StoragePersistence.Identity("guid:world-2"), "同名世界隔离");
    }
}}
