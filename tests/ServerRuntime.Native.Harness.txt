using System;
using System.IO;
using System.Linq;
using System.Reflection;
using HarmonyLib;

public static class ServerRuntimeTests
{
    private const BindingFlags Flags = BindingFlags.Static | BindingFlags.Instance | BindingFlags.Public | BindingFlags.NonPublic;
    public static int Main(string[] args)
    {
        try
        {
            string game = "D:/App/Game/Steam/steamapps/common/Terraria";
            Environment.SetEnvironmentVariable("TERRARIA_MODDER_DEDSERV", "1");
            Assembly server = null;
            AppDomain.CurrentDomain.AssemblyResolve += (sender, request) => {
                string name = new AssemblyName(request.Name).Name;
                var loaded = AppDomain.CurrentDomain.GetAssemblies().FirstOrDefault(a => a.GetName().Name == name);
                if (loaded != null) return loaded;
                if (name == "Terraria" && server != null) return server;
                foreach (string path in new[] { Path.Combine(game, name + ".exe"), Path.Combine(game, "TerrariaModder/core", name + ".dll"),
                    Path.Combine(game, "TerrariaModder/core/deps", name + ".dll"), Path.Combine(Environment.GetEnvironmentVariable("QOT_SOURCE_ROOT"), "lib", name + ".dll") })
                    if (File.Exists(path)) return Assembly.LoadFrom(path);
                return null;
            };
            server = Assembly.LoadFrom(Path.Combine(game, "TerrariaServer.exe"));
            server.GetType("Terraria.Program").GetField("SavePath", Flags).SetValue(null, Path.Combine(AppDomain.CurrentDomain.BaseDirectory, "test-save"));
            foreach (string name in server.GetManifestResourceNames().Where(n => n.EndsWith(".dll")))
                using (var stream = server.GetManifestResourceStream(name)) using (var buffer = new MemoryStream()) { stream.CopyTo(buffer); Assembly.Load(buffer.ToArray()); }
            server.GetType("Terraria.Main").GetField("dedServ", Flags).SetValue(null, true);
            server.GetType("Terraria.Main").GetField("netMode", Flags).SetValue(null, 2);
            var core = Assembly.LoadFrom(Path.Combine(game, "TerrariaModder/core/TerrariaModder.Core.dll"));
            var entry = Assembly.LoadFrom(Path.GetFullPath(args[0]));
            object config = Activator.CreateInstance(entry.GetType("QualityOfTerraria.QualityOfTerrariaConfig", true));
            object manifest = Activator.CreateInstance(core.GetType("TerrariaModder.Core.Manifest.ModManifest", true));
            manifest.GetType().GetProperty("Id").SetValue(manifest, "quality-of-terraria");
            Type contextType = core.GetType("TerrariaModder.Core.ModContext", true);
            object context = contextType.GetConstructors().Single().Invoke(new object[] { null, Path.GetDirectoryName(Path.GetFullPath(args[0])), manifest, config });
            object mod = Activator.CreateInstance(entry.GetType("QualityOfTerraria.Mod", true));
            mod.GetType().GetMethod("Initialize").Invoke(mod, new[] { context });
            object runtime = mod.GetType().GetField("_serverRuntime", Flags)?.GetValue(mod);
            Check(runtime != null, "服务器加载匹配 TerrariaServer 的运行模块");
            Assembly production = runtime.GetType().Assembly;
            Check(production.GetReferencedAssemblies().Any(a => a.Name == "TerrariaServer") && !production.GetReferencedAssemblies().Any(a => a.Name == "Terraria"), "服务器运行模块不引用客户端 Terraria");
            Check(ReferenceEquals(runtime.GetType().GetProperty("Config", Flags).GetValue(null), config), "服务器沿用加载器的配置实例和保存路径");
            mod.GetType().GetMethod("OnContentReady").Invoke(mod, new[] { context });
            var update = server.GetType("Terraria.Main", true).GetMethod("DoUpdate", Flags);
            Check(Harmony.GetPatchInfo(update)?.Owners.Contains("QualityOfTerraria.Server") == true, "服务器主线程更新挂接到真实服务器程序集");
            Check(runtime.GetType().GetMethod("OnWorldLoad") != null && runtime.GetType().GetMethod("OnWorldUnload") != null, "服务器运行模块暴露世界会话生命周期");
            Check(runtime.GetType().GetField("_serverRuntime", Flags) == null, "服务器模块不递归加载自身");
            Console.WriteLine("PASS 真实专服程序集加载、共享配置、服务器世界更新挂接与生命周期入口（未启动游戏）");
            return 0;
        }
        catch (Exception error) { Console.WriteLine(error); return 1; }
    }
    private static void Check(bool condition, string text) { if (!condition) throw new Exception("FAIL " + text); Console.WriteLine("PASS " + text); }
}
