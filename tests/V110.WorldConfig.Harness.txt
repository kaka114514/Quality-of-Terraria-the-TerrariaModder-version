using System;
using System.Collections.Generic;
using System.Reflection;
using QualityOfTerraria;
using TerrariaModder.Core.Config;
public static class ConfigChecks {
 static int failed; static void Check(bool ok,string name){Console.WriteLine((ok?"PASS ":"FAIL ")+name);if(!ok)failed++;}
 static PropertyInfo P(string name){return typeof(QualityOfTerrariaConfig).GetProperty(name);}
 static void Range(string key,int min,int max,object expected){var p=P(key); Check(p!=null,key+" 存在");if(p==null)return;var c=new QualityOfTerrariaConfig();Check(p.GetValue(c).Equals(expected),key+" 默认值");var r=(RangeAttribute)Attribute.GetCustomAttribute(p,typeof(RangeAttribute));Check(r!=null&&r.Min==min&&r.Max==max,key+" 范围");foreach(var meta in c.GetPropertyMetadata()){if(meta.Key!=key)continue;meta.SetValue(c,min-1);Check(Convert.ToDouble(p.GetValue(c))==min,key+" Core下限");meta.SetValue(c,max+1);Check(Convert.ToDouble(p.GetValue(c))==max,key+" Core上限");}}
 static void Apply(QualityOfTerrariaConfig c,Dictionary<string,object> raw){typeof(ModConfig).Assembly.GetType("TerrariaModder.Core.Config.ConfigManager").GetMethod("ApplyRawToProperties",BindingFlags.Static|BindingFlags.NonPublic).Invoke(null,new object[]{c,raw});typeof(ModConfig).GetMethod("ApplyMigration",BindingFlags.Instance|BindingFlags.Public|BindingFlags.NonPublic).Invoke(c,new object[]{raw,1});}
 static void Migration(bool enabled,int expected){var c=new QualityOfTerrariaConfig();Apply(c,new Dictionary<string,object>{{"ExtraTileReach",enabled},{"UniversalPylons",enabled}});Check(P("ExtraTileReach").GetValue(c).Equals(expected),"旧触及布尔 "+enabled+" → "+expected);foreach(string key in new[]{"PylonPlacementUnrestricted","PylonTeleportWithoutNearbyPlayer","PylonTeleportIgnoreDanger","PylonTeleportIgnoreBiome"}){var p=P(key);Check(p!=null&&p.GetValue(c).Equals(enabled),"旧晶塔迁移 "+key);}}
 public static int Main(){Range("ExtraTileReach",0,30,5);Range("RedPotionRequirement",1,300,20);Range("CoinDropMultiplier",1,25,1);Range("BannerKillRequirement",1,500,50);Range("SpawnRateMultiplier",0,100,1f);Check(P("BossQuickRespawn")!=null,"BOSS复活独立开关");foreach(string key in new[]{"MaxStack9999","QuestFishStack","NoDamageVariance","SpawnRateMultiplier","NoEvilSpread","UniversalPylons","PylonPlacementUnrestricted","PylonTeleportWithoutNearbyPlayer","PylonTeleportIgnoreDanger","PylonTeleportIgnoreBiome","CoinDropMultiplier","BannerKillRequirement","RefreshTravelShop"}){var p=P(key);Check(p!=null&&Attribute.IsDefined(p,typeof(ServerAttribute)),key+" 服务器配置");}Migration(true,5);Migration(false,0);var preserve=new QualityOfTerrariaConfig();Apply(preserve,new Dictionary<string,object>{{"ExtraTileReach",29},{"UniversalPylons",true},{"PylonTeleportIgnoreBiome",false}});Check(preserve.ExtraTileReach==29&&!preserve.PylonTeleportIgnoreBiome,"迁移保留整数触及和显式独立开关");return failed;}
}
