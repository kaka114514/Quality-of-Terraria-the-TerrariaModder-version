using System;
using System.IO;
using System.Linq;
using System.Reflection;
public static class StartupLoadOrder {
    public static int Main(string[] args) {
        string gameDir = "D:/App/Game/Steam/steamapps/common/Terraria";
        string deps = Path.Combine(gameDir, "TerrariaModder/core/deps");
        string phase = "mod-discovery";
        int failures = 0;
        AppDomain.CurrentDomain.AssemblyResolve += (sender, request) => {
            Assembly loaded = AppDomain.CurrentDomain.GetAssemblies().FirstOrDefault(a => a.FullName == request.Name);
            if (loaded != null) return loaded;
            string file = new AssemblyName(request.Name).Name.ToLowerInvariant() + ".dll";
            if (new AssemblyName(request.Name).Name == "Terraria") return Assembly.UnsafeLoadFrom(Path.Combine(gameDir, "Terraria.exe"));
            foreach (string folder in new[] { gameDir, Path.Combine(gameDir, "TerrariaModder"), deps, Path.Combine(gameDir, "TerrariaModder/mods") }) {
                string path = Path.Combine(folder, file);
                if (File.Exists(path)) return Assembly.UnsafeLoadFrom(path);
            }
            Console.WriteLine("UNRESOLVED " + phase + " " + request.Name + " requested-by=" + request.RequestingAssembly);
            return null;
        };
        foreach (string path in Directory.GetFiles(deps, "*.dll")) Assembly.UnsafeLoadFrom(path);
        Assembly.UnsafeLoadFrom(Path.Combine(gameDir, "TerrariaModder/core/TerrariaModder.Core.dll"));
        Assembly mod = Assembly.UnsafeLoadFrom(Path.GetFullPath(args[0]));
        foreach (Type type in mod.GetTypes()) {
            foreach (string name in new[] { "Init", "Initialize" }) {
                try {
                    MethodInfo method = type.GetMethod(name);
                    if (method == null || method.GetParameters().Length != 0) continue;
                    Console.WriteLine("EARLY-CALL " + type.FullName + "." + name);
                    method.Invoke(new object(), new object[0]);
                } catch (Exception error) {
                    failures++;
                    Console.WriteLine("FAIL early initialization " + type.FullName + " " + error);
                }
            }
        }
        phase = "embedded-dependencies";
        Assembly game = Assembly.UnsafeLoadFrom(Path.Combine(gameDir, "Terraria.exe"));
        game.GetType("Terraria.Program").GetField("SavePath").SetValue(null, Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.MyDocuments), "My Games/Terraria"));
        foreach (string resource in game.GetManifestResourceNames().Where(n => n.Contains(".dll"))) {
            using (Stream stream = game.GetManifestResourceStream(resource))
            using (MemoryStream buffer = new MemoryStream()) {
                stream.CopyTo(buffer);
                Assembly.Load(buffer.ToArray());
            }
        }
        phase = "game-main";
        try {
            game.GetType("Terraria.Main").GetField("gameMenu").GetValue(null);
            Console.WriteLine("PASS Main static initialization after embedded dependency load");
        } catch (Exception error) {
            failures++;
            Console.WriteLine("FAIL game-main " + error);
        }
        return failures == 0 ? 0 : 1;
    }
}
