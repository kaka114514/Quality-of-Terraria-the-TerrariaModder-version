using System;
using System.IO;
using System.Reflection;
using HarmonyLib;
using QualityOfTerraria.Features;
using QualityOfTerraria.Patches;
public static class NativeHooks {
 public static int Main() {
  string game="D:/App/Game/Steam/steamapps/common/Terraria";
  AppDomain.CurrentDomain.AssemblyResolve+=(sender,args)=>{
   string name=new AssemblyName(args.Name).Name+".dll";
   foreach(string path in new[]{Path.Combine(game,name),Path.Combine(game,"TerrariaModder/core",name),Path.Combine(game,"TerrariaModder/core/deps",name),"E:/Work/泰拉瑞亚更好的体验移植/lib/"+name}) if(File.Exists(path))return Assembly.LoadFrom(path);
   return null;
  };
  var harmony=new Harmony("qot.world.native-check");int checks=0,failed=0;
  foreach(Type patch in new[]{typeof(WorldQoLCompletionPatches.SpawnRatePatch),typeof(WorldQoLCompletionPatches.ZeroSpawnRatePatch),typeof(WorldQoLCompletionPatches.TreeDepthPatch),typeof(WorldQoLCompletionPatches.TreeGrowthPatch),typeof(WorldQoLCompletionPatches.LifeFruitGrowthPatch),typeof(WorldQoLCompletionPatches.EvilHousingPatch),typeof(WorldQoLCompletionPatches.BombPlayerDamagePatch),typeof(Autofisher.FishingStoragePatch)}) {
   checks++;
   try { var methods=harmony.CreateClassProcessor(patch).Patch(); if(patch==typeof(WorldQoLCompletionPatches.TreeGrowthPatch)&&methods.Count!=2)throw new Exception("树木速生必须同时挂接地表与地下更新，实际："+methods.Count); Console.WriteLine("PASS "+patch.Name+" "+string.Join(", ",methods)); }
   catch(Exception e){failed++;Console.WriteLine("FAIL "+patch.Name+" "+e);}
  }
  Console.WriteLine("真实1.4.5.8 IL挂接检查："+(checks-failed)+"/"+checks+" 通过（未启动游戏）");return failed;
 }
}
