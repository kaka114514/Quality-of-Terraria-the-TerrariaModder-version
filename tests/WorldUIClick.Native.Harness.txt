using System;
using System.IO;
using System.Reflection;
using System.Runtime.Serialization;
using HarmonyLib;
using Microsoft.Xna.Framework.Input;

public static class WorldUIClickNative
{
    private static readonly BindingFlags Static = BindingFlags.Static | BindingFlags.Public | BindingFlags.NonPublic;
    private static bool _physicalDown;
    private static int _checks;
    private static Type _main, _mod, _renderer;
    private static MethodInfo _logic;

    public static int Main()
    {
        try
        {
            string game = "D:/App/Game/Steam/steamapps/common/Terraria";
            var terraria = Assembly.LoadFrom(Path.Combine(game, "Terraria.exe"));
            AppDomain.CurrentDomain.AssemblyResolve += (sender, args) =>
            {
                string name = new AssemblyName(args.Name).Name;
                if (name == "Terraria") return terraria;
                foreach (string path in new[] {
                    Path.Combine(AppDomain.CurrentDomain.BaseDirectory, name + ".dll"),
                    Path.Combine(game, name + ".dll"),
                    Path.Combine(game, "TerrariaModder/core", name + ".dll"),
                    Path.Combine(game, "TerrariaModder/core/deps", name + ".dll"),
                    Path.Combine(Environment.GetEnvironmentVariable("QOT_SOURCE_ROOT"), "lib", name + ".dll") })
                    if (File.Exists(path)) return Assembly.LoadFrom(path);
                foreach (string resource in terraria.GetManifestResourceNames())
                    if (resource.EndsWith("." + name + ".dll", StringComparison.OrdinalIgnoreCase))
                        using (var stream = terraria.GetManifestResourceStream(resource))
                        using (var buffer = new MemoryStream())
                        { stream.CopyTo(buffer); return Assembly.Load(buffer.ToArray()); }
                return null;
            };
            var program = terraria.GetType("Terraria.Program", true);
            program.GetField("SavePath", Static).SetValue(null, Path.Combine(AppDomain.CurrentDomain.BaseDirectory, "test-save"));
            _main = terraria.GetType("Terraria.Main", true);
            var playerType = terraria.GetType("Terraria.Player", true);
            var player = FormatterServices.GetUninitializedObject(playerType);
            var players = Array.CreateInstance(playerType, 1);
            players.SetValue(player, 0);
            Field("player", players); Field("myPlayer", 0);
            Field("gameMenu", false); Field("dedServ", false);
            Field("playerInventory", true); Field("ingameOptionsWindow", false);

            var production = Assembly.LoadFrom(Path.Combine(AppDomain.CurrentDomain.BaseDirectory, "QualityOfTerraria.dll"));
            _mod = production.GetType("QualityOfTerraria.Mod", true);
            _mod.GetProperty("WorldActive", Static).GetSetMethod(true).Invoke(null, new object[] { true });
            var core = Assembly.LoadFrom(Path.Combine(game, "TerrariaModder/core/TerrariaModder.Core.dll"));
            _renderer = core.GetType("TerrariaModder.Core.UI.UIRenderer", true);
            _logic = _main.GetMethod("DrawInterface_0_InterfaceLogic1", Static);

            var environment = new Harmony("qot.world-ui-click.native.environment");
            environment.Patch(playerType.GetProperty("selectedItem").GetGetMethod(), new HarmonyMethod(typeof(WorldUIClickNative), nameof(SelectedSlot)));
            environment.Patch(playerType.GetProperty("UsingOrReusingItem").GetGetMethod(), new HarmonyMethod(typeof(WorldUIClickNative), nameof(UsingItem)));
            environment.Patch(typeof(Mouse).GetMethod("GetState", Type.EmptyTypes), new HarmonyMethod(typeof(WorldUIClickNative), nameof(PhysicalMouse)));
            Frame(true);
            Check(!Get<bool>("mouseLeftRelease"), "原版鼠标槽使用物品确实清除左键 release，复现故障前提");

            var fix = new Harmony("qot.world-ui-click.native.production");
            var patched = fix.CreateClassProcessor(production.GetType("QualityOfTerraria.Patches.WorldUIClickPatch", true)).Patch();
            Check(patched.Count == 1 && Harmony.GetPatchInfo(_logic).Owners.Contains(fix.Id), "实际生产补丁挂接真实 Terraria 1.4.5.8 世界界面入口");
            Frame(false); Frame(true);
            Check(Get<bool>("mouseLeft") && Get<bool>("mouseLeftRelease"), "打开原版背包时恢复真实左键新按下事件");
            Check((bool)_renderer.GetProperty("MouseLeftClick", Static).GetValue(null), "真实 Core UIRenderer.MouseLeftClick 接收到点击");
            Frame(true);
            Check(!Get<bool>("mouseLeftRelease"), "持续按住下一帧不重复点击");
            Field("playerInventory", false);
            Frame(false); Frame(true);
            Check(!Get<bool>("mouseLeftRelease"), "全部界面关闭时保留原版攻击 release 判定");
            Field("ingameOptionsWindow", true);
            Frame(false); Frame(true);
            Check(Get<bool>("mouseLeftRelease"), "原版世界内设置界面恢复点击");
            Field("ingameOptionsWindow", false);
            core.GetType("TerrariaModder.Core.UI.ModMenu", true).GetField("_visible", Static).SetValue(null, true);
            Frame(false); Frame(true);
            Check(Get<bool>("mouseLeftRelease"), "加载器 F6 菜单恢复点击");
            core.GetType("TerrariaModder.Core.UI.ModMenu", true).GetField("_visible", Static).SetValue(null, false);
            foreach (string panel in new[] { "ExtremeStorageUI", "SuperVaultUI", "PlayerStatsUI", "BuffAndSpawnRateUI", "StructureLocatorUI" })
            {
                var panelType = production.GetType("QualityOfTerraria.UI." + panel, true);
                SetVisible(panelType, true);
                Frame(false); Frame(true);
                Check(Get<bool>("mouseLeftRelease"), "本模组 " + panel + " 可见时恢复点击");
                SetVisible(panelType, false);
            }
            Field("playerInventory", true); Field("gameMenu", true);
            Frame(false); Frame(true);
            Check(!Get<bool>("mouseLeftRelease"), "菜单阶段不插入世界点击事件");
            Field("gameMenu", false); Field("dedServ", true);
            Frame(false); Frame(true);
            Check(!Get<bool>("mouseLeftRelease"), "专用服务器不插入点击事件");
            Console.WriteLine("真实 net48 x86 世界界面点击检查通过：" + _checks + " 项（未启动游戏）");
            return 0;
        }
        catch (Exception error) { Console.WriteLine(error); return 1; }
    }

    public static bool SelectedSlot(ref int __result) { __result = 58; return false; }
    public static bool UsingItem(ref bool __result) { __result = true; return false; }
    public static bool PhysicalMouse(ref MouseState __result)
    {
        __result = new MouseState(100, 100, 0, _physicalDown ? ButtonState.Pressed : ButtonState.Released,
            ButtonState.Released, ButtonState.Released, ButtonState.Released, ButtonState.Released);
        return false;
    }
    private static void Frame(bool down)
    {
        _physicalDown = down;
        Field("mouseLeft", down); Field("mouseLeftRelease", true);
        _logic.Invoke(null, null);
    }
    private static void Field(string name, object value) { _main.GetField(name, Static).SetValue(null, value); }
    private static void SetVisible(Type panel, bool visible)
    {
        var property = panel.GetProperty("IsOpen", Static);
        if (property != null) property.SetValue(null, visible);
        else panel.GetField("IsOpen", Static).SetValue(null, visible);
    }
    private static T Get<T>(string name) { return (T)_main.GetField(name, Static).GetValue(null); }
    private static void Check(bool condition, string message)
    { if (!condition) throw new Exception("FAIL " + message); _checks++; Console.WriteLine("PASS " + message); }
}
